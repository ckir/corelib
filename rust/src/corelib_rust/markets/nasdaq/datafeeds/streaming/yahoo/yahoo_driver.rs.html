13/// Length-only debug note for frames the mapper rejects. Takes ONLY the byte count —
14/// it is structurally impossible to leak the (base64-protobuf) frame contents.
15pub(crate) fn note_undecodable(provider: &str, bytes: usize) {
16    tracing::debug!(target: "corelib_rust::stream", provider, bytes, "undecodable frame skipped");
17}

88/// Decode one Yahoo WS text frame into the raw payload + zero/one/two unified events.
89/// Returns `None` only when the frame is not a decodable pricing envelope (skip frame).
90/// The raw `JsPricingData` is ALWAYS present when `Some` (so `on_pricing` always fires),
91/// even when the uni vec is empty (heartbeats / all-zero-price messages).
92pub fn parse_yahoo_message(text: &str, source: &str) -> Option<(JsPricingData, Vec<MarketEvent>)> {
93    let obj: serde_json::Value = serde_json::from_str(text).ok()?;
94    let b64 = obj.get("message").and_then(|m| m.as_str())?;
95    let pricing = decode_yahoo_message(b64).ok()?;
96    let raw = JsPricingData::from(pricing);
97
98    let ticker = raw.id.clone();
99    let timestamp = ts_from_millis(raw.time);
100    let mut uni: Vec<MarketEvent> = Vec::new();
101
102    // Trade then Quote (finstream order).
103    if raw.price > 0.0 {
104        uni.push(MarketEvent::Trade {
105            source: source.to_string(),
106            data: Trade {
107                ticker: ticker.clone(),
108                timestamp,
109                price: raw.price,
110                extras: TradeExtras::Yahoo(trade_extras(&raw)),
111                raw: None,
112            },
113        });
114    }
115    if raw.bid_price > 0.0 || raw.ask_price > 0.0 {
116        uni.push(MarketEvent::Quote {
117            source: source.to_string(),
118            data: Quote {
119                ticker,
120                timestamp,
121                price: (raw.bid_price + raw.ask_price) / 2.0,
122                extras: QuoteExtras::Yahoo(quote_extras(&raw)),
123                raw: None,
124            },
125        });
126    }
127    Some((raw, uni))
128}

142const DEFAULT_YAHOO_WS_URL: &str = "wss://streamer.finance.yahoo.com/?version=2";
143const PING_INTERVAL: u64 = 30;
144
145/// One Yahoo real-time stream connection. The supervisor owns the reconnect loop; this driver
146/// performs a single attempt (connect → subscribe → pump). Yahoo is unauthenticated and
147/// single-channel; the redb DB (shared `Arc<Database>` from the host) is the single source of
148/// truth for subscriptions, read fresh (bare keys) on every (re)connect.
149pub struct YahooDriver {
150    pub name: String,
151    pub base_url: Option<String>,
152    pub silence_seconds: u32,
153    pub db: Arc<Database>,
154    pub table: &'static str,
155}
156
157impl YahooDriver {
158    /// Read the FULL persisted bare-key subscription set from redb. Called on every (re)connect.
159    pub fn load_subscriptions(&self) -> Vec<String> {
160        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
161        let mut out = Vec::new();
162        if let Ok(rtx) = self.db.begin_read() {
163            if let Ok(t) = rtx.open_table(table) {
164                if let Ok(iter) = t.iter() {
165                    for item in iter.flatten() {
166                        out.push(item.0.value().to_string());
167                    }
168                }
169            }
170        }
171        out
172    }
173}

175impl ProviderDriver for YahooDriver {
176    fn validate(&self) -> Result<(), String> {
177        Ok(()) // Yahoo is unauthenticated
178    }
179
180    fn connect_once<'a>(
181        &'a self,
182        symbols: &'a [String], // unused: Yahoo resumes from redb (load_subscriptions), not this snapshot
183        tx: &'a mpsc::Sender<CoreEvent>,
184        sub_rx: &'a mut mpsc::Receiver<SubRequest>,
185        stop_rx: &'a mut mpsc::Receiver<()>,
186    ) -> BoxFuture<'a, AttemptOutcome> {
187        async move {
188            let _ = symbols; // resume is read fresh from redb each attempt (reconnect-safe)
189            let url = self
190                .base_url
191                .clone()
192                .unwrap_or_else(|| DEFAULT_YAHOO_WS_URL.to_string());
193            let (mut ws, _) = match connect_async(&url).await {
194                Ok(v) => v,
195                Err(_) => return AttemptOutcome::NeverConnected,
196            };
197            let _ = tx
198                .send(CoreEvent::Status(ProviderStatus::Connected {
199                    provider: ProviderKind::Yahoo,
200                }))
201                .await;
202
203            // initial subscribe: full persisted bare-key set (single-channel)
204            let subs = self.load_subscriptions();
205            if !subs.is_empty() {
206                let payload = serde_json::json!({ "subscribe": subs }).to_string();
207                if ws.send(Message::Text(payload.into())).await.is_err() {
208                    return AttemptOutcome::ConnectedThenDropped;
209                }
210            }
211
212            let mut silence = tokio::time::interval(tokio::time::Duration::from_secs(
213                self.silence_seconds.max(1) as u64,
214            ));
215            let mut ping = tokio::time::interval(tokio::time::Duration::from_secs(PING_INTERVAL));
216            let _ = silence.tick().await;
217            let _ = ping.tick().await;
218
219            loop {
220                tokio::select! {
221                    _ = stop_rx.recv() => return AttemptOutcome::Stopped,
222                    req = sub_rx.recv() => {
223                        match req {
224                            // single-channel: ignore the channel tag, subscribe the bare symbols
225                            Some(SubRequest { symbols: syms, .. }) => {
226                                tracing::trace!(target: "corelib_rust::stream", symbols = syms.len(), "sub request");
227                                if !syms.is_empty() {
228                                    let payload = serde_json::json!({ "subscribe": syms }).to_string();
229                                    let _ = ws.send(Message::Text(payload.into())).await;
230                                }
231                            }
232                            None => return AttemptOutcome::Stopped,
233                        }
234                    }
235                    _ = ping.tick() => { let _ = ws.send(Message::Ping(vec![].into())).await; }
236                    _ = silence.tick() => return AttemptOutcome::ConnectedThenDropped,
237                    msg = ws.next() => match msg {
238                        Some(Ok(Message::Text(t))) => {
239                            silence.reset();
240                            if let Some((raw, uni)) = parse_yahoo_message(&t, &self.name) {
241                                let _ = tx.send(CoreEvent::Pricing { raw: RawPricing::Yahoo(raw), uni }).await;
242                            } else {
243                                note_undecodable("yahoo", t.len());
244                            }
245                        }
246                        Some(Ok(Message::Ping(p))) => { let _ = ws.send(Message::Pong(p)).await; }
247                        Some(Ok(_)) => {}
248                        Some(Err(_)) | None => return AttemptOutcome::ConnectedThenDropped,
249                    }
250                }
251            }
252        }
253        .boxed()
254    }
255}

495        if let Ok((stream, _)) = listener.accept().await {
496            let mut ws = accept_async(stream).await.expect("ws accept");
497            let p = PricingData {

507            let env = serde_json::json!({ "type": "pricing", "message": b64 }).to_string();
508            ws.send(text(&env)).await.unwrap();
509            tokio::time::sleep(Duration::from_millis(300)).await;
510        }

517        tokio::spawn(serve_yahoo(listener));

526        let (tx, mut rx) = mpsc::channel::<CoreEvent>(64);
527        let (_sub_tx, mut sub_rx) = mpsc::channel::<SubRequest>(8);
528        let (_stop_tx, mut stop_rx) = mpsc::channel::<()>(1);
529
530        let connect = driver.connect_once(&[] as &[String], &tx, &mut sub_rx, &mut stop_rx);

536                    _ = &mut connect => return None,
537                    ev = rx.recv() => match ev {
538                        Some(CoreEvent::Pricing { raw: RawPricing::Yahoo(p), .. }) => return Some(p),
539                        Some(_) => continue, // skip Status(Connected)
540                        None => return None,
541                    },
542                }
543            }
544        })
545        .await
546        .expect("timed out waiting for pricing");
547
548        let p = got.expect("driver ended before delivering a pricing event");
549        assert_eq!(p.id, "AAPL");
550        assert_eq!(p.price, 191.5);
551    }

561    #[derive(Clone)]
562    struct VecWriter(Arc<Mutex<Vec<u8>>>);
563    impl Write for VecWriter {
564        fn write(&mut self, b: &[u8]) -> std::io::Result<usize> {
565            self.0.lock().unwrap().extend_from_slice(b);
566            Ok(b.len())
567        }
568        fn flush(&mut self) -> std::io::Result<()> {
569            Ok(())
570        }
571    }
572    impl<'a> MakeWriter<'a> for VecWriter {
573        type Writer = VecWriter;
574        fn make_writer(&'a self) -> Self::Writer {
575            self.clone()
576        }
577    }
