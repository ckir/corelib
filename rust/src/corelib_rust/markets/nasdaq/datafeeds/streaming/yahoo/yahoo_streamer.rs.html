32/// Configuration parameters for the Yahoo price streamer. Yahoo has no credentials; `db_path`
33/// and `base_url` are retained for API/back-compat and test injection.
34#[napi(object)]
35#[derive(Clone, Debug, Serialize, Deserialize)]
36pub struct YahooConfig {
37    /// Legacy no-op: the redb path is derived per-instance by the host (or via the `YAHOO_DB`
38    /// env override). Retained for back-compat; ignored.
39    pub db_path: Option<String>,
40    /// Threshold in seconds for silence detection before triggering a reconnect (default 60).
41    pub silence_seconds: Option<u32>,
42    /// Optional override for the WebSocket URL (defaults to the Yahoo v2 streamer).
43    pub base_url: Option<String>,
44}

85/// N-API facade for Yahoo real-time streaming (dual-mode: raw + unified).
86///
87/// `ThreadsafeFunction` is not `Clone`; each is wrapped in `Arc` so the pump closure can hold a
88/// cheap reference-counted copy. `on_market_event` is optional (absent ⇒ raw-only consumers).
89#[napi]
90pub struct YahooStreaming {
91    inner: Arc<Mutex<YahooInner>>,
92    on_log: Arc<ThreadsafeFunction<LogRecord>>,
93    on_pricing: Arc<BoundedTsfn<JsPricingData>>,
94    on_event: Arc<ThreadsafeFunction<EventRecord>>,
95    on_market_event: Option<Arc<BoundedTsfn<String>>>,
96}
97
98#[napi]
99impl YahooStreaming {
100    /// Constructs a new `YahooStreaming` with the JS callback functions.
101    /// Order: (on_log, on_pricing, on_event, [on_market_event]). `on_market_event` is optional —
102    /// pass it to also receive the unified `"market"` stream.
103    #[napi(constructor)]
104    pub fn new(
105        on_log: ThreadsafeFunction<LogRecord>,
106        on_pricing: ThreadsafeFunction<
107            JsPricingData,
108            Unknown<'static>,
109            JsPricingData,
110            Status,
111            true,
112            false,
113            1024,
114        >,
115        on_event: ThreadsafeFunction<EventRecord>,
116        on_market_event: Option<
117            ThreadsafeFunction<String, Unknown<'static>, String, Status, true, false, 1024>,
118        >,
119    ) -> napi::Result<Self> {
120        let host = WebsocketStreamerHost::new(
121            unique_db_path("yahoo_streaming", "YAHOO_DB"),
122            "yahoo_subscriptions",
123            "yahoo".into(),
124            ProviderKind::Yahoo,
125        )
126        .map_err(|e| napi::Error::from_reason(e.to_string()))?;
127        Ok(Self {
128            inner: Arc::new(Mutex::new(YahooInner {
129                host,
130                config: YahooConfig::default_empty(),
131                started: false,
132            })),
133            on_log: Arc::new(on_log),
134            on_pricing: Arc::new(on_pricing),
135            on_event: Arc::new(on_event),
136            on_market_event: on_market_event.map(Arc::new),
137        })
138    }
139
140    /// Set config (silence threshold / optional base_url override).
141    #[napi]
142    pub async fn init(&self, config: YahooConfig) -> Result<()> {
143        self.inner.lock().await.config = config;
144        Ok(())
145    }
146
147    /// Start streaming; the driver resumes persisted subscriptions (redb) on every (re)connect.
148    #[napi]
149    pub async fn start(&self) -> Result<()> {
150        let mut g = self.inner.lock().await;
151        if g.started {
152            return Ok(());
153        }
154        let driver = YahooDriver {
155            name: "yahoo".into(),
156            base_url: g.config.base_url.clone(),
157            silence_seconds: g.config.silence_seconds.unwrap_or(60),
158            db: g.host.db_handle(),
159            table: g.host.table_name(),
160        };
161        let on_pricing = Arc::clone(&self.on_pricing);
162        let on_event = Arc::clone(&self.on_event);
163        let on_market = self.on_market_event.clone();
164        g.host.start(
165            driver,
166            Vec::new(),
167            ReconnectPolicy {
168                jitter: true,
169                ..Default::default()
170            },
171            move |ev: CoreEvent| match ev {
172                CoreEvent::Pricing {
173                    raw: RawPricing::Yahoo(p),
174                    uni,
175                } => {
176                    let _ = on_pricing.call(Ok(p), ThreadsafeFunctionCallMode::NonBlocking);
177                    if let Some(cb) = on_market.as_ref() {
178                        for u in &uni {
179                            if let Ok(j) = serde_json::to_string(u) {
180                                let _ = cb.call(Ok(j), ThreadsafeFunctionCallMode::NonBlocking);
181                            }
182                        }
183                    }
184                }
185                CoreEvent::Status(status) => {
186                    let (t, d) = match status {
187                        ProviderStatus::Connected { .. } => ("connected".to_string(), None),
188                        ProviderStatus::Disconnected { reason, .. } => {
189                            ("disconnected".to_string(), Some(reason))
190                        }
191                        ProviderStatus::Reconnecting {
192                            attempt, delay_ms, ..
193                        } => (
194                            "reconnecting".to_string(),
195                            Some(format!("attempt {attempt}, {delay_ms}ms")),
196                        ),
197                        ProviderStatus::Error { message, .. } => {
198                            ("error".to_string(), Some(message))
199                        }
200                    };
201                    let _ = on_event.call(
202                        Ok(EventRecord { r#type: t, data: d }),
203                        ThreadsafeFunctionCallMode::NonBlocking,
204                    );
205                }
206                // RawPricing::Alpaca / ::Finnhub never reach the Yahoo pump.
207                _ => {}
208            },
209        );
210        let _ = self.on_log.call(
211            Ok(LogRecord {
212                level: "debug".into(),
213                msg: "yahoo start".into(),
214                extras: None,
215            }),
216            ThreadsafeFunctionCallMode::NonBlocking,
217        );
218        g.started = true;
219        Ok(())
220    }
221
222    /// Subscribe to additional symbols (bare list; persisted to redb + sent live to the driver).
223    #[napi]
224    pub async fn subscribe(&self, symbols: Vec<String>) -> Result<()> {
225        self.inner.lock().await.host.subscribe(symbols).await;
226        Ok(())
227    }
228
229    /// Remove `symbols` from the persisted set (so they don't resume on restart).
230    #[napi]
231    pub async fn unsubscribe(&self, symbols: Vec<String>) -> Result<()> {
232        self.inner.lock().await.host.unsubscribe(symbols).await;
233        Ok(())
234    }
235
236    /// Clears all persisted subscriptions and stops the stream.
237    #[napi]
238    pub async fn clean(&self) -> Result<()> {
239        let mut g = self.inner.lock().await;
240        let _ = g.host.delete_subscriptions_table();
241        if let Some(tx) = g.host.stop_tx.take() {
242            let _ = tx.try_send(());
243        }
244        g.started = false;
245        Ok(())
246    }
247
248    /// Gracefully stops the streaming supervisor (keeps persisted subscriptions for resume).
249    #[napi]
250    pub async fn stop(&self) -> Result<()> {
251        let mut g = self.inner.lock().await;
252        if let Some(tx) = g.host.stop_tx.take() {
253            let _ = tx.try_send(());
254        }
255        g.started = false;
256        Ok(())
257    }
258}

267    fn fresh_host() -> WebsocketStreamerHost {
268        WebsocketStreamerHost::new(
269            unique_db_path("yahoo_streaming", "YAHOO_DB_TEST_UNSET"),
270            "yahoo_subscriptions",
271            "yahoo".into(),
272            ProviderKind::Yahoo,
273        )
274        .expect("test host")
275    }
276
277    #[tokio::test]
278    async fn subscribe_persists_bare_keys_and_resumes() {
279        let host = fresh_host();
280        host.subscribe(vec!["AAPL".into(), "BTC-USD".into()]).await;
281        let mut got = host.get_persisted_subscriptions();
282        got.sort();
283        assert_eq!(got, vec!["AAPL".to_string(), "BTC-USD".to_string()]);
284    }
285
286    #[tokio::test]
287    async fn unsubscribe_removes_persisted_key() {
288        let host = fresh_host();
289        host.subscribe(vec!["AAPL".into(), "MSFT".into()]).await;
290        host.unsubscribe(vec!["AAPL".into()]).await;
291        assert_eq!(host.get_persisted_subscriptions(), vec!["MSFT".to_string()]);
292    }
