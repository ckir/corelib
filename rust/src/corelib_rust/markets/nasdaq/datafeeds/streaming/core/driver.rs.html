1//! The single-attempt provider driver contract. The supervisor owns the reconnect loop.
2use crate::markets::nasdaq::datafeeds::streaming::core::types::CoreEvent;
3use tokio::sync::mpsc;
4
5/// A live subscription update routed to a driver: a channel name + the symbols.
6/// Single-channel providers (Finnhub) use a fixed channel and ignore the tag; multi-channel
7/// providers (Alpaca: trades/quotes/bars) route by `channel`.
8#[allow(dead_code)] // consumed by drivers in later tasks
9#[derive(Debug, Clone)]
10pub struct SubRequest {
11    pub channel: String,
12    pub symbols: Vec<String>,
13}
14
15/// Outcome of one connection attempt, used by the supervisor to decide the next action.
16#[allow(dead_code)] // used by drivers/supervisor in later tasks
17#[derive(Debug, Clone, PartialEq, Eq)]
18pub enum AttemptOutcome {
19    /// Connected (authenticated/received data) then the socket dropped → reset backoff, retry.
20    ConnectedThenDropped,
21    /// Never connected/authenticated this attempt → grow backoff, retry.
22    NeverConnected,
23    /// Fatal error (e.g. auth rejected) → stop the supervisor.
24    Fatal(String),
25    /// Graceful stop requested via stop_rx → stop the supervisor.
26    Stopped,
27}
28
29/// A self-contained driver for ONE financial data provider connection attempt.
30///
31/// `connect_once` returns a **`BoxFuture`** (not a bare `async fn`): a stable-Rust `async fn` in a
32/// trait yields a future that is NOT guaranteed `Send`, so awaiting it inside the generic
33/// `run_supervisor<D>` (Task 7) that is `tokio::spawn`-ed (Task 8) would FAIL to compile. `BoxFuture`
34/// pins it as `Send` explicitly. *(agy plan-pass 🔴 [Structural].)*
35#[allow(dead_code)] // used by drivers/supervisor in later tasks
36pub trait ProviderDriver: Send + Sync + 'static {
37    /// Validate config (keys present, etc.) before the first attempt.
38    fn validate(&self) -> Result<(), String> {
39        Ok(())
40    }
41
42    /// Perform ONE connection attempt: connect, (auth), subscribe `symbols`, apply live
43    /// `sub_rx` updates, push `CoreEvent`s to `tx` (including `Status::Connected` on success),
44    /// and resolve when the socket drops, a fatal error occurs, or `stop_rx` fires.
45    fn connect_once<'a>(
46        &'a self,
47        symbols: &'a [String],
48        tx: &'a mpsc::Sender<CoreEvent>,
49        sub_rx: &'a mut mpsc::Receiver<SubRequest>,
50        stop_rx: &'a mut mpsc::Receiver<()>,
51    ) -> futures::future::BoxFuture<'a, AttemptOutcome>;
52}
