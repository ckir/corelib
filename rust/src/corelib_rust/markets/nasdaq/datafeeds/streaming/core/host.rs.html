13static INSTANCE_COUNTER: AtomicU64 = AtomicU64::new(0);

15/// Channel tag used by the single-channel `subscribe()` path. Single-channel drivers (Finnhub)
16/// ignore it; it exists only so the live `SubRequest` is well-formed.
17const DEFAULT_SUB_CHANNEL: &str = "default";
18
19/// Build a unique per-instance redb path under temp_dir (b-1: no shared-file lock).
20pub fn unique_db_path(prefix: &str, env_override: &str) -> std::path::PathBuf {
21    if let Ok(p) = std::env::var(env_override) {
22        return std::path::PathBuf::from(p);
23    }
24    let pid = std::process::id();
25    let seq = INSTANCE_COUNTER.fetch_add(1, Ordering::Relaxed);
26    let nanos = std::time::SystemTime::now()
27        .duration_since(std::time::UNIX_EPOCH)
28        .unwrap_or_default()
29        .subsec_nanos();
30    std::env::temp_dir().join(format!("{prefix}_{pid}_{seq}_{nanos}.redb"))
31}

33/// Shared coordination state. `Pump` consumes MarketEvents and calls the provider's typed TSFNs.
34/// Like the b-1 Alpaca fix, we store the *monitor* task (which owns/awaits the supervisor) and the
35/// pump task; the supervisor itself exits via `stop_tx`.
36pub struct WebsocketStreamerHost {
37    pub(crate) db: Arc<Database>,
38    pub(crate) table: &'static str,
39    #[allow(dead_code)]
40    // retained for instance identification / future logging; events tag via ProviderKind
41    pub(crate) source: String,
42    pub(crate) provider: ProviderKind, // generic: NOT hardcoded (agy plan-pass 🔴) — used in panic events
43    pub(crate) sub_tx: Option<mpsc::Sender<SubRequest>>,
44    pub(crate) stop_tx: Option<mpsc::Sender<()>>,
45    pub(crate) monitor_task: Option<JoinHandle<()>>,
46    pub(crate) pump_task: Option<JoinHandle<()>>,
47}

49impl WebsocketStreamerHost {
50    /// Open/create the per-instance redb file. `source` names the instance; `provider` tags events.
51    pub fn new(
52        db_path: std::path::PathBuf,
53        table: &'static str,
54        source: String,
55        provider: ProviderKind,
56    ) -> Self {
57        let db = Arc::new(Database::create(&db_path).expect("Failed to open redb"));
58        Self {
59            db,
60            table,
61            source,
62            provider,
63            sub_tx: None,
64            stop_tx: None,
65            monitor_task: None,
66            pump_task: None,
67        }
68    }

70    /// Load previously-persisted subscription tickers so a restarted instance resumes them
71    /// (agy plan-pass 🔴 — without this, keeping redb buys nothing). Merge into `start`'s symbols.
72    pub fn get_persisted_subscriptions(&self) -> Vec<String> {
73        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
74        let mut subs = Vec::new();
75        if let Ok(rtx) = self.db.begin_read() {
76            if let Ok(t) = rtx.open_table(table) {
77                let Ok(iter) = t.iter() else { return subs };
78                for item in iter {
79                    let Ok(entry) = item else { continue };
80                    subs.push(entry.0.value().to_string());
81                }
82            }
83        }
84        subs
85    }

87    /// Start the supervisor (driving `driver`), a panic monitor, and a pump that forwards events.
88    pub fn start<D, P>(
89        &mut self,
90        driver: D,
91        symbols: Vec<String>,
92        policy: ReconnectPolicy,
93        mut on_event_pump: P,
94    ) where
95        D: ProviderDriver,
96        P: FnMut(CoreEvent) + Send + 'static,
97    {
98        let (tx, mut rx) = mpsc::channel::<CoreEvent>(1024);
99        let (sub_tx, sub_rx) = mpsc::channel::<SubRequest>(64);
100        let (stop_tx, stop_rx) = mpsc::channel::<()>(1);
101        self.sub_tx = Some(sub_tx);
102        self.stop_tx = Some(stop_tx);
103
104        // b-1 §3.4: monitor awaits the supervisor; on panic emit a synthetic Error event via a tx clone.
105        // CoreEvent::Status carries the provider tag (not source), so the monitor only needs `provider`.
106        let monitor_tx = tx.clone();
107        let provider = self.provider; // ProviderKind is Copy — generic, not hardcoded
108
109        let sup = tokio::spawn(run_supervisor(driver, symbols, tx, sub_rx, stop_rx, policy));
110
111        self.monitor_task = Some(tokio::spawn(async move {
112            if let Err(e) = sup.await {
113                if e.is_panic() {
114                    let _ = monitor_tx
115                        .send(CoreEvent::Status(ProviderStatus::Error {
116                            provider,
117                            message: "supervisor task panicked; stream is dead".into(),
118                        }))
119                        .await;
120                }
121            }
122        }));
123
124        self.pump_task = Some(tokio::spawn(async move {
125            while let Some(ev) = rx.recv().await {
126                on_event_pump(ev);
127            }
128        }));
129    }

131    /// Persist + push a subscription set to the live driver.
132    pub async fn subscribe(&self, symbols: Vec<String>) {
133        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
134        if let Ok(wtx) = self.db.begin_write() {
135            if let Ok(mut t) = wtx.open_table(table) {
136                for s in &symbols {
137                    let _ = t.insert(s.as_str(), true);
138                }
139            }
140            let _ = wtx.commit();
141        }
142        if let Some(tx) = &self.sub_tx {
143            // Single-channel callers (Finnhub) persist bare keys and ignore the channel tag;
144            // multi-channel providers (Alpaca, Task 4+) use `subscribe_channel` instead.
145            let _ = tx
146                .send(SubRequest {
147                    channel: DEFAULT_SUB_CHANNEL.to_string(),
148                    symbols,
149                })
150                .await;
151        }
152    }

154    /// Remove `symbols` from the persisted subscription set so they do not resurrect on restart.
155    /// (Phase 1: live WS-unsubscribe through the driver is deferred to Phase 2; this keeps redb
156    /// consistent with developer intent — agy convergent 🟢.)
157    pub async fn unsubscribe(&self, symbols: Vec<String>) {
158        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
159        if let Ok(wtx) = self.db.begin_write() {
160            if let Ok(mut t) = wtx.open_table(table) {
161                for s in &symbols {
162                    let _ = t.remove(s.as_str());
163                }
164            }
165            let _ = wtx.commit();
166        }
167    }

169    /// Write `channel:symbol` composite keys (multi-channel providers).
170    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
171    pub fn subscribe_channel(&self, channel: &str, symbols: Vec<String>) {
172        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
173        if let Ok(wtx) = self.db.begin_write() {
174            if let Ok(mut t) = wtx.open_table(table) {
175                for s in &symbols {
176                    let _ = t.insert(format!("{channel}:{s}").as_str(), true);
177                }
178            }
179            let _ = wtx.commit();
180        }
181    }

183    /// Send a live per-channel `SubRequest` to the running driver (immediate in-session effect).
184    /// No-op before `start()` (when `sub_tx` is `None`): resume is still safe because the caller
185    /// also persisted via `subscribe_channel`, and the driver reads redb fresh on every connect.
186    #[allow(dead_code)] // consumed by Alpaca in Task 8
187    pub fn subscribe_channel_live(&self, channel: &str, symbols: Vec<String>) {
188        if let Some(tx) = &self.sub_tx {
189            let _ = tx.try_send(SubRequest {
190                channel: channel.to_string(),
191                symbols,
192            });
193        }
194    }

210    /// Read symbols for `target_channel`; a colon-less (legacy bare) key is treated as `default_channel`.
211    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
212    pub fn get_persisted_subscriptions_for_channel(
213        &self,
214        target_channel: &str,
215        default_channel: &str,
216    ) -> Vec<String> {
217        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
218        let mut out = Vec::new();
219        if let Ok(rtx) = self.db.begin_read() {
220            if let Ok(t) = rtx.open_table(table) {
221                let Ok(iter) = t.iter() else { return out };
222                for item in iter {
223                    let Ok(entry) = item else { continue };
224                    let key = entry.0.value().to_string();
225                    let (ch, sym) = key
226                        .split_once(':')
227                        .unwrap_or((default_channel, key.as_str()));
228                    if ch == target_channel {
229                        out.push(sym.to_string());
230                    }
231                }
232            }
233        }
234        out
235    }

237    /// Drop the entire subscriptions table (for `clean()`).
238    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
239    pub fn delete_subscriptions_table(&self) -> Result<(), String> {
240        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
241        let wtx = self.db.begin_write().map_err(|e| e.to_string())?;
242        let _ = wtx.delete_table(table);
243        wtx.commit().map_err(|e| e.to_string())
244    }

246    /// Cheap clone of the shared redb handle (drivers read persisted subscriptions directly).
247    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
248    pub fn db_handle(&self) -> Arc<Database> {
249        Arc::clone(&self.db)
250    }

252    /// The subscriptions table name (for driver-side reads).
253    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
254    pub fn table_name(&self) -> &'static str {
255        self.table
256    }

259impl Drop for WebsocketStreamerHost {
260    fn drop(&mut self) {
261        // Supervisor exits gracefully via stop_tx (mirrors b-1 Alpaca). Abort monitor + pump.
262        if let Some(tx) = self.stop_tx.take() {
263            let _ = tx.try_send(());
264        }
265        if let Some(t) = self.monitor_task.take() {
266            t.abort(); // b-1: abort monitor task
267        }
268        if let Some(t) = self.pump_task.take() {
269            t.abort(); // b-1: abort BOTH facade tasks
270        }
271    }
272}

274#[cfg(test)]
275mod host_persistence_tests {
276    use super::*;
277    fn fresh() -> WebsocketStreamerHost {
278        WebsocketStreamerHost::new(
279            unique_db_path("test_host", "TEST_HOST_DB_UNSET"),
280            "test_subscriptions",
281            "test".into(),
282            ProviderKind::Alpaca,
283        )
284    }
285    #[tokio::test]
286    async fn channel_keys_roundtrip_and_unsubscribe_is_precise() {
287        let h = fresh();
288        h.subscribe_channel("quotes", vec!["AAPL".into(), "MSFT".into()]);
289        h.subscribe_channel("trades", vec!["AAPL".into()]);
290        let mut q = h.get_persisted_subscriptions_for_channel("quotes", "quotes");
291        q.sort();
292        assert_eq!(q, vec!["AAPL".to_string(), "MSFT".to_string()]);
293        assert_eq!(
294            h.get_persisted_subscriptions_for_channel("trades", "quotes"),
295            vec!["AAPL".to_string()]
296        );
297        h.unsubscribe_channel("quotes", vec!["AAPL".into()]);
298        assert_eq!(
299            h.get_persisted_subscriptions_for_channel("quotes", "quotes"),
300            vec!["MSFT".to_string()]
301        );
302        // trades:AAPL untouched
303        assert_eq!(
304            h.get_persisted_subscriptions_for_channel("trades", "quotes"),
305            vec!["AAPL".to_string()]
306        );
307    }
308    #[tokio::test]
309    async fn colonless_legacy_key_defaults_to_channel() {
310        let h = fresh();
311        // simulate a legacy bare-symbol record via the existing API
312        h.subscribe(vec!["TSLA".into()]).await; // writes "TSLA" (bare) through the legacy path
313        let got = h.get_persisted_subscriptions_for_channel("quotes", "quotes");
314        assert!(got.contains(&"TSLA".to_string()));
315    }
316    #[tokio::test]
317    async fn delete_subscriptions_table_clears_all() {
318        let h = fresh();
319        h.subscribe_channel("quotes", vec!["AAPL".into()]);
320        h.delete_subscriptions_table().unwrap();
321        assert!(h
322            .get_persisted_subscriptions_for_channel("quotes", "quotes")
323            .is_empty());
324    }
