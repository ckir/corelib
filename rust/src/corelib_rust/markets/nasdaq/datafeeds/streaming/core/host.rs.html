13static INSTANCE_COUNTER: AtomicU64 = AtomicU64::new(0);

15/// Channel tag used by the single-channel `subscribe()` path. Single-channel drivers (Finnhub)
16/// ignore it; it exists only so the live `SubRequest` is well-formed.
17const DEFAULT_SUB_CHANNEL: &str = "default";
18
19/// Build a unique per-instance redb path under temp_dir (b-1: no shared-file lock).
20pub fn unique_db_path(prefix: &str, env_override: &str) -> std::path::PathBuf {
21    if let Ok(p) = std::env::var(env_override) {
22        return std::path::PathBuf::from(p);
23    }
24    let pid = std::process::id();
25    let seq = INSTANCE_COUNTER.fetch_add(1, Ordering::Relaxed);
26    let nanos = std::time::SystemTime::now()
27        .duration_since(std::time::UNIX_EPOCH)
28        .unwrap_or_default()
29        .subsec_nanos();
30    std::env::temp_dir().join(format!("{prefix}_{pid}_{seq}_{nanos}.redb"))
31}

33/// Shared coordination state. `Pump` consumes MarketEvents and calls the provider's typed TSFNs.
34/// Like the b-1 Alpaca fix, we store the *monitor* task (which owns/awaits the supervisor) and the
35/// pump task; the supervisor itself exits via `stop_tx`.
36pub struct WebsocketStreamerHost {
37    pub(crate) db: Arc<Database>,
38    pub(crate) table: &'static str,
39    #[allow(dead_code)]
40    // retained for instance identification / future logging; events tag via ProviderKind
41    pub(crate) source: String,
42    pub(crate) provider: ProviderKind, // generic: NOT hardcoded (agy plan-pass 🔴) — used in panic events
43    pub(crate) sub_tx: Option<mpsc::Sender<SubRequest>>,
44    pub(crate) stop_tx: Option<mpsc::Sender<()>>,
45    pub(crate) monitor_task: Option<JoinHandle<()>>,
46    pub(crate) pump_task: Option<JoinHandle<()>>,
47}

49#[derive(Debug)]
50pub enum HostError {
51    DbOpen(redb::DatabaseError),
52}
53impl std::fmt::Display for HostError {
54    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
55        match self {
56            HostError::DbOpen(e) => write!(f, "failed to open redb: {e}"),
57        }
58    }
59}
60impl std::error::Error for HostError {
61    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
62        match self {
63            HostError::DbOpen(e) => Some(e),
64        }
65    }
66}

68impl WebsocketStreamerHost {
69    /// Open/create the per-instance redb file. `source` names the instance; `provider` tags events.
70    /// Fallible: returns `Err(HostError::DbOpen)` if redb cannot open the path (e.g. a shared path
71    /// already held open by another instance) — callers must handle it instead of aborting.
72    pub fn new(
73        db_path: std::path::PathBuf,
74        table: &'static str,
75        source: String,
76        provider: ProviderKind,
77    ) -> Result<Self, HostError> {
78        let db = Arc::new(Database::create(&db_path).map_err(HostError::DbOpen)?);
79        Ok(Self {
80            db,
81            table,
82            source,
83            provider,
84            sub_tx: None,
85            stop_tx: None,
86            monitor_task: None,
87            pump_task: None,
88        })
89    }
90
91    /// Load previously-persisted subscription tickers so a restarted instance resumes them
92    /// (agy plan-pass 🔴 — without this, keeping redb buys nothing). Merge into `start`'s symbols.
93    pub fn get_persisted_subscriptions(&self) -> Vec<String> {
94        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
95        let mut subs = Vec::new();
96        if let Ok(rtx) = self.db.begin_read() {
97            if let Ok(t) = rtx.open_table(table) {
98                let Ok(iter) = t.iter() else { return subs };
99                for item in iter {
100                    let Ok(entry) = item else { continue };
101                    subs.push(entry.0.value().to_string());
102                }
103            }
104        }
105        subs
106    }

108    /// Start the supervisor (driving `driver`), a panic monitor, and a pump that forwards events.
109    pub fn start<D, P>(
110        &mut self,
111        driver: D,
112        symbols: Vec<String>,
113        policy: ReconnectPolicy,
114        mut on_event_pump: P,
115    ) where
116        D: ProviderDriver,
117        P: FnMut(CoreEvent) + Send + 'static,
118    {
119        let (tx, mut rx) = mpsc::channel::<CoreEvent>(1024);
120        let (sub_tx, sub_rx) = mpsc::channel::<SubRequest>(64);
121        let (stop_tx, stop_rx) = mpsc::channel::<()>(1);
122        self.sub_tx = Some(sub_tx);
123        self.stop_tx = Some(stop_tx);
124
125        // b-1 §3.4: monitor awaits the supervisor; on panic emit a synthetic Error event via a tx clone.
126        // CoreEvent::Status carries the provider tag (not source), so the monitor only needs `provider`.
127        let monitor_tx = tx.clone();
128        let provider = self.provider; // ProviderKind is Copy — generic, not hardcoded
129
130        let sup = tokio::spawn(run_supervisor(driver, symbols, tx, sub_rx, stop_rx, policy));
131
132        self.monitor_task = Some(tokio::spawn(async move {
133            if let Err(e) = sup.await {
134                if e.is_panic() {
135                    let _ = monitor_tx
136                        .send(CoreEvent::Status(ProviderStatus::Error {
137                            provider,
138                            message: "supervisor task panicked; stream is dead".into(),
139                        }))
140                        .await;
141                }
142            }
143        }));
144
145        self.pump_task = Some(tokio::spawn(async move {
146            while let Some(ev) = rx.recv().await {
147                on_event_pump(ev);
148            }
149        }));
150    }

152    /// Persist + push a subscription set to the live driver.
153    pub async fn subscribe(&self, symbols: Vec<String>) {
154        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
155        if let Ok(wtx) = self.db.begin_write() {
156            if let Ok(mut t) = wtx.open_table(table) {
157                for s in &symbols {
158                    let _ = t.insert(s.as_str(), true);
159                }
160            }
161            let _ = wtx.commit();
162        }
163        if let Some(tx) = &self.sub_tx {
164            // Single-channel callers (Finnhub) persist bare keys and ignore the channel tag;
165            // multi-channel providers (Alpaca, Task 4+) use `subscribe_channel` instead.
166            let _ = tx
167                .send(SubRequest {
168                    channel: DEFAULT_SUB_CHANNEL.to_string(),
169                    symbols,
170                })
171                .await;
172        }
173    }
174
175    /// Remove `symbols` from the persisted subscription set so they do not resurrect on restart.
176    /// (Phase 1: live WS-unsubscribe through the driver is deferred to Phase 2; this keeps redb
177    /// consistent with developer intent — agy convergent 🟢.)
178    pub async fn unsubscribe(&self, symbols: Vec<String>) {
179        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
180        if let Ok(wtx) = self.db.begin_write() {
181            if let Ok(mut t) = wtx.open_table(table) {
182                for s in &symbols {
183                    let _ = t.remove(s.as_str());
184                }
185            }
186            let _ = wtx.commit();
187        }
188    }

190    /// Write `channel:symbol` composite keys (multi-channel providers).
191    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
192    pub fn subscribe_channel(&self, channel: &str, symbols: Vec<String>) {
193        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
194        if let Ok(wtx) = self.db.begin_write() {
195            if let Ok(mut t) = wtx.open_table(table) {
196                for s in &symbols {
197                    let _ = t.insert(format!("{channel}:{s}").as_str(), true);
198                }
199            }
200            let _ = wtx.commit();
201        }
202    }

204    /// Send a live per-channel `SubRequest` to the running driver (immediate in-session effect).
205    /// No-op before `start()` (when `sub_tx` is `None`): resume is still safe because the caller
206    /// also persisted via `subscribe_channel`, and the driver reads redb fresh on every connect.
207    #[allow(dead_code)] // consumed by Alpaca in Task 8
208    pub fn subscribe_channel_live(&self, channel: &str, symbols: Vec<String>) {
209        if let Some(tx) = &self.sub_tx {
210            let _ = tx.try_send(SubRequest {
211                channel: channel.to_string(),
212                symbols,
213            });
214        }
215    }

231    /// Read symbols for `target_channel`; a colon-less (legacy bare) key is treated as `default_channel`.
232    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
233    pub fn get_persisted_subscriptions_for_channel(
234        &self,
235        target_channel: &str,
236        default_channel: &str,
237    ) -> Vec<String> {
238        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
239        let mut out = Vec::new();
240        if let Ok(rtx) = self.db.begin_read() {
241            if let Ok(t) = rtx.open_table(table) {
242                let Ok(iter) = t.iter() else { return out };
243                for item in iter {
244                    let Ok(entry) = item else { continue };
245                    let key = entry.0.value().to_string();
246                    let (ch, sym) = key
247                        .split_once(':')
248                        .unwrap_or((default_channel, key.as_str()));
249                    if ch == target_channel {
250                        out.push(sym.to_string());
251                    }
252                }
253            }
254        }
255        out
256    }

258    /// Drop the entire subscriptions table (for `clean()`).
259    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
260    pub fn delete_subscriptions_table(&self) -> Result<(), String> {
261        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
262        let wtx = self.db.begin_write().map_err(|e| e.to_string())?;
263        let _ = wtx.delete_table(table);
264        wtx.commit().map_err(|e| e.to_string())
265    }

267    /// Cheap clone of the shared redb handle (drivers read persisted subscriptions directly).
268    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
269    pub fn db_handle(&self) -> Arc<Database> {
270        Arc::clone(&self.db)
271    }

273    /// The subscriptions table name (for driver-side reads).
274    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
275    pub fn table_name(&self) -> &'static str {
276        self.table
277    }

280impl Drop for WebsocketStreamerHost {
281    fn drop(&mut self) {
282        // Supervisor exits gracefully via stop_tx (mirrors b-1 Alpaca). Abort monitor + pump.
283        if let Some(tx) = self.stop_tx.take() {
284            let _ = tx.try_send(());
285        }
286        if let Some(t) = self.monitor_task.take() {
287            t.abort(); // b-1: abort monitor task
288        }
289        if let Some(t) = self.pump_task.take() {
290            t.abort(); // b-1: abort BOTH facade tasks
291        }
292    }
293}

295#[cfg(test)]
296mod host_persistence_tests {
297    use super::*;
298    fn fresh() -> WebsocketStreamerHost {
299        WebsocketStreamerHost::new(
300            unique_db_path("test_host", "TEST_HOST_DB_UNSET"),
301            "test_subscriptions",
302            "test".into(),
303            ProviderKind::Alpaca,
304        )
305        .expect("test host")
306    }
307    #[tokio::test]
308    async fn channel_keys_roundtrip_and_unsubscribe_is_precise() {
309        let h = fresh();
310        h.subscribe_channel("quotes", vec!["AAPL".into(), "MSFT".into()]);
311        h.subscribe_channel("trades", vec!["AAPL".into()]);
312        let mut q = h.get_persisted_subscriptions_for_channel("quotes", "quotes");
313        q.sort();
314        assert_eq!(q, vec!["AAPL".to_string(), "MSFT".to_string()]);
315        assert_eq!(
316            h.get_persisted_subscriptions_for_channel("trades", "quotes"),
317            vec!["AAPL".to_string()]
318        );
319        h.unsubscribe_channel("quotes", vec!["AAPL".into()]);
320        assert_eq!(
321            h.get_persisted_subscriptions_for_channel("quotes", "quotes"),
322            vec!["MSFT".to_string()]
323        );
324        // trades:AAPL untouched
325        assert_eq!(
326            h.get_persisted_subscriptions_for_channel("trades", "quotes"),
327            vec!["AAPL".to_string()]
328        );
329    }
330    #[tokio::test]
331    async fn colonless_legacy_key_defaults_to_channel() {
332        let h = fresh();
333        // simulate a legacy bare-symbol record via the existing API
334        h.subscribe(vec!["TSLA".into()]).await; // writes "TSLA" (bare) through the legacy path
335        let got = h.get_persisted_subscriptions_for_channel("quotes", "quotes");
336        assert!(got.contains(&"TSLA".to_string()));
337    }
338    #[tokio::test]
339    async fn delete_subscriptions_table_clears_all() {
340        let h = fresh();
341        h.subscribe_channel("quotes", vec!["AAPL".into()]);
342        h.delete_subscriptions_table().unwrap();
343        assert!(h
344            .get_persisted_subscriptions_for_channel("quotes", "quotes")
345            .is_empty());
346    }
