13static INSTANCE_COUNTER: AtomicU64 = AtomicU64::new(0);

15/// Channel tag used by the single-channel `subscribe()` path. Single-channel drivers (Finnhub)
16/// ignore it; it exists only so the live `SubRequest` is well-formed.
17const DEFAULT_SUB_CHANNEL: &str = "default";

19/// Build a unique per-instance redb path under temp_dir (b-1: no shared-file lock).
20pub fn unique_db_path(prefix: &str, env_override: &str) -> std::path::PathBuf {
21    if let Ok(p) = std::env::var(env_override) {
22        return std::path::PathBuf::from(p);
23    }
24    let pid = std::process::id();
25    let seq = INSTANCE_COUNTER.fetch_add(1, Ordering::Relaxed);
26    let nanos = std::time::SystemTime::now()
27        .duration_since(std::time::UNIX_EPOCH)
28        .unwrap_or_default()
29        .subsec_nanos();
30    std::env::temp_dir().join(format!("{prefix}_{pid}_{seq}_{nanos}.redb"))
31}

33/// Shared coordination state. `Pump` consumes MarketEvents and calls the provider's typed TSFNs.
34/// Like the b-1 Alpaca fix, we store the *monitor* task (which owns/awaits the supervisor) and the
35/// pump task; the supervisor itself exits via `stop_tx`.
36pub struct WebsocketStreamerHost {
37    pub(crate) db: Arc<Database>,
38    pub(crate) table: &'static str,
39    #[allow(dead_code)]
40    // retained for instance identification / future logging; events tag via ProviderKind
41    pub(crate) source: String,
42    pub(crate) provider: ProviderKind, // generic: NOT hardcoded (agy plan-pass 🔴) — used in panic events
43    pub(crate) sub_tx: Option<mpsc::Sender<SubRequest>>,
44    pub(crate) stop_tx: Option<mpsc::Sender<()>>,
45    pub(crate) monitor_task: Option<JoinHandle<()>>,
46    pub(crate) pump_task: Option<JoinHandle<()>>,
47}

49#[derive(Debug)]
50pub enum HostError {
51    DbOpen(redb::DatabaseError),
52}
53impl std::fmt::Display for HostError {
54    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
55        match self {
56            HostError::DbOpen(e) => write!(f, "failed to open redb: {e}"),
57        }
58    }
59}
60impl std::error::Error for HostError {
61    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
62        match self {
63            HostError::DbOpen(e) => Some(e),
64        }
65    }
66}

68/// Map a CoreEvent to a short static kind string for hot-path tracing (REDACTION:
69/// kind only — never the raw pricing payload).
70fn core_event_kind(ev: &CoreEvent) -> &'static str {
71    match ev {
72        CoreEvent::Pricing { .. } => "pricing",
73        CoreEvent::Status(_) => "status",
74    }
75}

77impl WebsocketStreamerHost {
78    /// Open/create the per-instance redb file. `source` names the instance; `provider` tags events.
79    /// Fallible: returns `Err(HostError::DbOpen)` if redb cannot open the path (e.g. a shared path
80    /// already held open by another instance) — callers must handle it instead of aborting.
81    pub fn new(
82        db_path: std::path::PathBuf,
83        table: &'static str,
84        source: String,
85        provider: ProviderKind,
86    ) -> Result<Self, HostError> {
87        let db = Arc::new(Database::create(&db_path).map_err(HostError::DbOpen)?);
88        Ok(Self {
89            db,
90            table,
91            source,
92            provider,
93            sub_tx: None,
94            stop_tx: None,
95            monitor_task: None,
96            pump_task: None,
97        })
98    }

100    /// Load previously-persisted subscription tickers so a restarted instance resumes them
101    /// (agy plan-pass 🔴 — without this, keeping redb buys nothing). Merge into `start`'s symbols.
102    pub fn get_persisted_subscriptions(&self) -> Vec<String> {
103        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
104        let mut subs = Vec::new();
105        if let Ok(rtx) = self.db.begin_read() {
106            if let Ok(t) = rtx.open_table(table) {
107                let Ok(iter) = t.iter() else { return subs };
108                for item in iter {
109                    let Ok(entry) = item else { continue };
110                    subs.push(entry.0.value().to_string());
111                }
112            }
113        }
114        subs
115    }

117    /// Start the supervisor (driving `driver`), a panic monitor, and a pump that forwards events.
118    pub fn start<D, P>(
119        &mut self,
120        driver: D,
121        symbols: Vec<String>,
122        policy: ReconnectPolicy,
123        mut on_event_pump: P,
124    ) where
125        D: ProviderDriver,
126        P: FnMut(CoreEvent) + Send + 'static,
127    {
128        let (tx, mut rx) = mpsc::channel::<CoreEvent>(1024);
129        let (sub_tx, sub_rx) = mpsc::channel::<SubRequest>(64);
130        let (stop_tx, stop_rx) = mpsc::channel::<()>(1);
131        self.sub_tx = Some(sub_tx);
132        self.stop_tx = Some(stop_tx);
133
134        // b-1 §3.4: monitor awaits the supervisor; on panic emit a synthetic Error event via a tx clone.
135        // CoreEvent::Status carries the provider tag (not source), so the monitor only needs `provider`.
136        let monitor_tx = tx.clone();
137        let provider = self.provider; // ProviderKind is Copy — generic, not hardcoded
138
139        let sup = tokio::spawn(run_supervisor(driver, symbols, tx, sub_rx, stop_rx, policy));
140
141        self.monitor_task = Some(tokio::spawn(async move {
142            if let Err(e) = sup.await {
143                if e.is_panic() {
144                    let _ = monitor_tx
145                        .send(CoreEvent::Status(ProviderStatus::Error {
146                            provider,
147                            message: "supervisor task panicked; stream is dead".into(),
148                        }))
149                        .await;
150                }
151            }
152        }));
153
154        self.pump_task = Some(tokio::spawn(async move {
155            while let Some(ev) = rx.recv().await {
156                tracing::trace!(target: "corelib_rust::stream::tick", kind = core_event_kind(&ev), "pump event");
157                on_event_pump(ev);
158            }
159        }));
160    }

162    /// Persist + push a subscription set to the live driver.
163    pub async fn subscribe(&self, symbols: Vec<String>) {
164        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
165        if let Ok(wtx) = self.db.begin_write() {
166            if let Ok(mut t) = wtx.open_table(table) {
167                for s in &symbols {
168                    let _ = t.insert(s.as_str(), true);
169                }
170            }
171            let _ = wtx.commit();
172        }
173        if let Some(tx) = &self.sub_tx {
174            // Single-channel callers (Finnhub) persist bare keys and ignore the channel tag;
175            // multi-channel providers (Alpaca, Task 4+) use `subscribe_channel` instead.
176            let _ = tx
177                .send(SubRequest {
178                    channel: DEFAULT_SUB_CHANNEL.to_string(),
179                    symbols,
180                })
181                .await;
182        }
183    }

185    /// Remove `symbols` from the persisted subscription set so they do not resurrect on restart.
186    /// (Phase 1: live WS-unsubscribe through the driver is deferred to Phase 2; this keeps redb
187    /// consistent with developer intent — agy convergent 🟢.)
188    pub async fn unsubscribe(&self, symbols: Vec<String>) {
189        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
190        if let Ok(wtx) = self.db.begin_write() {
191            if let Ok(mut t) = wtx.open_table(table) {
192                for s in &symbols {
193                    let _ = t.remove(s.as_str());
194                }
195            }
196            let _ = wtx.commit();
197        }
198    }

200    /// Write `channel:symbol` composite keys (multi-channel providers).
201    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
202    pub fn subscribe_channel(&self, channel: &str, symbols: Vec<String>) {
203        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
204        if let Ok(wtx) = self.db.begin_write() {
205            if let Ok(mut t) = wtx.open_table(table) {
206                for s in &symbols {
207                    let _ = t.insert(format!("{channel}:{s}").as_str(), true);
208                }
209            }
210            let _ = wtx.commit();
211        }
212    }

214    /// Send a live per-channel `SubRequest` to the running driver (immediate in-session effect).
215    /// No-op before `start()` (when `sub_tx` is `None`): resume is still safe because the caller
216    /// also persisted via `subscribe_channel`, and the driver reads redb fresh on every connect.
217    #[allow(dead_code)] // consumed by Alpaca in Task 8
218    pub fn subscribe_channel_live(&self, channel: &str, symbols: Vec<String>) {
219        if let Some(tx) = &self.sub_tx {
220            let _ = tx.try_send(SubRequest {
221                channel: channel.to_string(),
222                symbols,
223            });
224        }
225    }

241    /// Read symbols for `target_channel`; a colon-less (legacy bare) key is treated as `default_channel`.
242    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
243    pub fn get_persisted_subscriptions_for_channel(
244        &self,
245        target_channel: &str,
246        default_channel: &str,
247    ) -> Vec<String> {
248        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
249        let mut out = Vec::new();
250        if let Ok(rtx) = self.db.begin_read() {
251            if let Ok(t) = rtx.open_table(table) {
252                let Ok(iter) = t.iter() else { return out };
253                for item in iter {
254                    let Ok(entry) = item else { continue };
255                    let key = entry.0.value().to_string();
256                    let (ch, sym) = key
257                        .split_once(':')
258                        .unwrap_or((default_channel, key.as_str()));
259                    if ch == target_channel {
260                        out.push(sym.to_string());
261                    }
262                }
263            }
264        }
265        out
266    }

268    /// Drop the entire subscriptions table (for `clean()`).
269    #[allow(dead_code)] // consumed by Alpaca in Tasks 7/8
270    pub fn delete_subscriptions_table(&self) -> Result<(), String> {
271        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
272        let wtx = self.db.begin_write().map_err(|e| e.to_string())?;
273        let _ = wtx.delete_table(table);
274        wtx.commit().map_err(|e| e.to_string())
275    }

290impl Drop for WebsocketStreamerHost {
291    fn drop(&mut self) {
292        // Supervisor exits gracefully via stop_tx (mirrors b-1 Alpaca). Abort monitor + pump.
293        if let Some(tx) = self.stop_tx.take() {
294            let _ = tx.try_send(());
295        }
296        if let Some(t) = self.monitor_task.take() {
297            t.abort(); // b-1: abort monitor task
298        }
299        if let Some(t) = self.pump_task.take() {
300            t.abort(); // b-1: abort BOTH facade tasks
301        }
302    }
303}

305#[cfg(test)]
306mod host_persistence_tests {
307    use super::*;
308    fn fresh() -> WebsocketStreamerHost {
309        WebsocketStreamerHost::new(
310            unique_db_path("test_host", "TEST_HOST_DB_UNSET"),
311            "test_subscriptions",
312            "test".into(),
313            ProviderKind::Alpaca,
314        )
315        .expect("test host")
316    }
317    #[tokio::test]
318    async fn channel_keys_roundtrip_and_unsubscribe_is_precise() {
319        let h = fresh();
320        h.subscribe_channel("quotes", vec!["AAPL".into(), "MSFT".into()]);
321        h.subscribe_channel("trades", vec!["AAPL".into()]);
322        let mut q = h.get_persisted_subscriptions_for_channel("quotes", "quotes");
323        q.sort();
324        assert_eq!(q, vec!["AAPL".to_string(), "MSFT".to_string()]);
325        assert_eq!(
326            h.get_persisted_subscriptions_for_channel("trades", "quotes"),
327            vec!["AAPL".to_string()]
328        );
329        h.unsubscribe_channel("quotes", vec!["AAPL".into()]);
330        assert_eq!(
331            h.get_persisted_subscriptions_for_channel("quotes", "quotes"),
332            vec!["MSFT".to_string()]
333        );
334        // trades:AAPL untouched
335        assert_eq!(
336            h.get_persisted_subscriptions_for_channel("trades", "quotes"),
337            vec!["AAPL".to_string()]
338        );
339    }
340    #[tokio::test]
341    async fn colonless_legacy_key_defaults_to_channel() {
342        let h = fresh();
343        // simulate a legacy bare-symbol record via the existing API
344        h.subscribe(vec!["TSLA".into()]).await; // writes "TSLA" (bare) through the legacy path
345        let got = h.get_persisted_subscriptions_for_channel("quotes", "quotes");
346        assert!(got.contains(&"TSLA".to_string()));
347    }
348    #[tokio::test]
349    async fn delete_subscriptions_table_clears_all() {
350        let h = fresh();
351        h.subscribe_channel("quotes", vec!["AAPL".into()]);
352        h.delete_subscriptions_table().unwrap();
353        assert!(h
354            .get_persisted_subscriptions_for_channel("quotes", "quotes")
355            .is_empty());
356    }
357}
