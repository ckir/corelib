7/// Parse one Finnhub WS text frame into zero or more Trade events.
8/// Finnhub trade frame: {"type":"trade","data":[{"s":sym,"p":price,"v":vol,"t":ms,"c":[..]}]}
9pub fn parse_finnhub_frame(text: &str, source: &str) -> Vec<MarketEvent> {
10    let obj: serde_json::Value = match serde_json::from_str(text) {
11        Ok(v) => v,
12        Err(_) => return vec![],
13    };
14    if obj["type"].as_str() != Some("trade") {
15        return vec![];
16    }
17    let data = match obj["data"].as_array() {
18        Some(a) => a,
19        None => return vec![],
20    };
21    let mut out = Vec::new();
22    for item in data {
23        let ticker = match item["s"].as_str() {
24            Some(s) => s.to_string(),
25            None => continue,
26        };
27        let price = match item["p"].as_f64() {
28            Some(p) => p,
29            None => continue,
30        };
31        let volume = item["v"].as_f64().unwrap_or(0.0);
32        let time_ms = item["t"].as_i64().unwrap_or(0);
33        let timestamp = Utc
34            .timestamp_millis_opt(time_ms)
35            .single()
36            .unwrap_or_else(Utc::now);
37        let conditions = item["c"]
38            .as_array()
39            .map(|a| {
40                a.iter()
41                    .filter_map(|v| v.as_str().map(str::to_owned))
42                    .collect()
43            })
44            .unwrap_or_default();
45        out.push(MarketEvent::Trade {
46            source: source.to_string(),
47            data: Trade {
48                ticker,
49                timestamp,
50                price,
51                extras: TradeExtras::Finnhub(FinnhubTradeExtras { volume, conditions }),
52                raw: Some(item.to_string()),
53            },
54        });
55    }
56    out
57}

72const FINNHUB_WS: &str = "wss://ws.finnhub.io";

74/// Build the Finnhub websocket connect URL. Uses `base_url` when provided, else the default endpoint.
75/// Scheme/query formatting is byte-identical to the legacy hardcoded form when `base_url` is None.
76fn finnhub_ws_url(base_url: Option<&str>, token: &str) -> String {
77    let base = base_url.unwrap_or(FINNHUB_WS);
78    format!("{base}?token={token}")
79}

89impl FinnhubDriver {
90    /// Read the FULL persisted bare-key subscription set from redb. Called on every (re)connect
91    /// so dynamic mid-session subscribes survive reconnects (single source of truth = redb).
92    pub fn load_subscriptions(&self) -> Vec<String> {
93        let table: TableDefinition<&str, bool> = TableDefinition::new(self.table);
94        let mut out = Vec::new();
95        if let Ok(rtx) = self.db.begin_read() {
96            if let Ok(t) = rtx.open_table(table) {
97                if let Ok(iter) = t.iter() {
98                    for item in iter.flatten() {
99                        out.push(item.0.value().to_string());
100                    }
101                }
102            }
103        }
104        out
105    }
106}

108impl ProviderDriver for FinnhubDriver {
109    fn validate(&self) -> Result<(), String> {
110        if self.token.is_empty() {
111            Err("FINNHUB_API_KEY / token is required".into())
112        } else {
113            Ok(())
114        }
115    }
116
117    fn connect_once<'a>(
118        &'a self,
119        symbols: &'a [String],
120        tx: &'a mpsc::Sender<CoreEvent>,
121        sub_rx: &'a mut mpsc::Receiver<SubRequest>,
122        stop_rx: &'a mut mpsc::Receiver<()>,
123    ) -> BoxFuture<'a, AttemptOutcome> {
124        async move {
125            let url = finnhub_ws_url(self.base_url.as_deref(), &self.token);
126            let (mut ws, _) = match connect_async(&url).await {
127                Ok(v) => v,
128                Err(_) => return AttemptOutcome::NeverConnected,
129            };
130            // signal the supervisor to reset backoff
131            let _ = tx.send(CoreEvent::Status(ProviderStatus::Connected { provider: ProviderKind::Finnhub })).await;
132            let _ = symbols; // resume is read fresh from redb each attempt (reconnect-safe)
133            let mut current: Vec<String> = self.load_subscriptions();
134            for s in &current {
135                let m = serde_json::json!({ "type": "subscribe", "symbol": s }).to_string();
136                if ws.send(Message::Text(m.into())).await.is_err() { return AttemptOutcome::ConnectedThenDropped; }
137            }
138            loop {
139                tokio::select! {
140                    _ = stop_rx.recv() => return AttemptOutcome::Stopped,
141                    upd = sub_rx.recv() => {
142                        match upd {
143                            // Finnhub is single-channel: ignore `channel`, subscribe each new symbol.
144                            Some(SubRequest { symbols: syms, .. }) => {
145                                tracing::trace!(target: "corelib_rust::stream", symbols = syms.len(), "sub request");
146                                for s in &syms { if !current.contains(s) {
147                                    let m = serde_json::json!({ "type": "subscribe", "symbol": s }).to_string();
148                                    let _ = ws.send(Message::Text(m.into())).await; current.push(s.clone());
149                                }}
150                            }
151                            // sub_tx dropped → channel closed; recv() would return None forever and
152                            // busy-spin the select. End the attempt cleanly. (agy convergent 🟡)
153                            None => return AttemptOutcome::Stopped,
154                        }
155                    }
156                    msg = ws.next() => match msg {
157                        Some(Ok(Message::Text(t))) => {
158                            // dual-mode: each trade carries the raw FinnhubPricingData + the unified MarketEvent.
159                            for ev in parse_finnhub_frame(&t, &self.name) {
160                                if let Some(raw) = market_event_to_finnhub_pricing(&ev) {
161                                    let _ = tx.send(CoreEvent::Pricing { raw: RawPricing::Finnhub(raw), uni: vec![ev] }).await;
162                                }
163                            }
164                        }
165                        Some(Ok(Message::Ping(p))) => { let _ = ws.send(Message::Pong(p)).await; }
166                        Some(Ok(_)) => {}
167                        Some(Err(_)) | None => return AttemptOutcome::ConnectedThenDropped,
168                    }
169                }
170            }
171        }.boxed()
172    }
173}

228        {

230            {
231                let mut tab = w.open_table(t).unwrap();
232                tab.insert("AAPL", true).unwrap();
233                tab.insert("MSFT", true).unwrap();
234            }
235            w.commit().unwrap();
236        }
237        let d = FinnhubDriver {
238            token: "tok".into(),
239            name: "finnhub_main".into(),
240            base_url: None,
241            db: Arc::clone(&db),
242            table: "finnhub_subscriptions",
243        };
244        let mut subs = d.load_subscriptions();
245        subs.sort();
246        assert_eq!(subs, vec!["AAPL".to_string(), "MSFT".to_string()]);
247        // unsubscribe MSFT (committed) must NOT resurrect on the next read
248        {
249            let w = db.begin_write().unwrap();
250            {
251                let mut tab = w.open_table(t).unwrap();
252                tab.remove("MSFT").unwrap();
253            }
254            w.commit().unwrap();
255        }
256        assert_eq!(d.load_subscriptions(), vec!["AAPL".to_string()]);
257    }

275    /// Minimal Finnhub WS server: (read one subscribe) → one trade frame. No auth, no handshake.
276    async fn serve_finnhub(listener: TcpListener) {
277        if let Ok((stream, _)) = listener.accept().await {
278            let mut ws = accept_async(stream).await.expect("ws accept");
279            let _ = ws.next().await; // subscribe frame from driver
280            ws.send(text(
281                r#"{"type":"trade","data":[{"s":"AAPL","p":191.5,"v":100,"t":1700000000000,"c":["@"]}]}"#,
282            ))
283            .await
284            .unwrap();
285            tokio::time::sleep(Duration::from_millis(300)).await;
286        }
287    }

327                    ev = rx.recv() => match ev {

356    /// Collects the `symbol` of every Finnhub `{"type":"subscribe","symbol":..}` frame.
357    async fn serve_collect(listener: TcpListener, collected: Arc<Mutex<Vec<String>>>) {
358        if let Ok((stream, _)) = listener.accept().await {
359            let mut ws = accept_async(stream).await.unwrap();
360            let deadline = tokio::time::sleep(Duration::from_millis(800));
361            tokio::pin!(deadline);
362            loop {
363                tokio::select! {
364                    _ = &mut deadline => break,
365                    msg = ws.next() => match msg {
366                        Some(Ok(Message::Text(t))) => {
367                            if let Ok(v) = serde_json::from_str::<serde_json::Value>(&t) {
368                                if v["type"] == "subscribe" {
369                                    if let Some(s) = v["symbol"].as_str() {
370                                        collected.lock().unwrap().push(s.to_string());
371                                    }
372                                }
373                            }
374                        }
375                        Some(Ok(_)) => {}
376                        Some(Err(_)) | None => break,
377                    }
378                }
379            }
380        }
381    }

383    #[tokio::test]
384    async fn reconnect_after_unsubscribe_does_not_resurrect() {
385        let path = std::env::temp_dir().join(format!(
386            "test_finnhub_resurrect_{}_{}.redb",
387            std::process::id(),
388            std::time::SystemTime::now()
389                .duration_since(std::time::UNIX_EPOCH)
390                .unwrap()
391                .as_nanos()
392        ));
393        let db = Arc::new(redb::Database::create(path).unwrap());
394        let tbl: redb::TableDefinition<&str, bool> =
395            redb::TableDefinition::new("finnhub_subscriptions");
396        // seed A + B, then unsubscribe B (committed delete) BEFORE the (re)connect.
397        {
398            let w = db.begin_write().unwrap();
399            {
400                let mut tab = w.open_table(tbl).unwrap();
401                tab.insert("AAPL", true).unwrap();
402                tab.insert("MSFT", true).unwrap();
403            }
404            w.commit().unwrap();
405        }
406        {
407            let w = db.begin_write().unwrap();
408            {
409                let mut tab = w.open_table(tbl).unwrap();
410                tab.remove("MSFT").unwrap();
411            }
412            w.commit().unwrap();
413        }
414        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
415        let port = listener.local_addr().unwrap().port();
416        let collected = Arc::new(Mutex::new(Vec::<String>::new()));
417        tokio::spawn(serve_collect(listener, Arc::clone(&collected)));
418
419        let driver = FinnhubDriver {
420            token: "tok".into(),
421            name: "finnhub_main".into(),
422            base_url: Some(format!("ws://127.0.0.1:{port}/")),
423            db: Arc::clone(&db),
424            table: "finnhub_subscriptions",
425        };
426        let (tx, _rx) = mpsc::channel::<CoreEvent>(64); // keep _rx alive so sends don't fail
427        let (_sub_tx, mut sub_rx) = mpsc::channel::<SubRequest>(8);
428        let (stop_tx, mut stop_rx) = mpsc::channel::<()>(1);
429        let connect = driver.connect_once(&[] as &[String], &tx, &mut sub_rx, &mut stop_rx);
430        tokio::pin!(connect);
431        // let it connect + send its initial subscribe, then stop the attempt.
432        tokio::select! {
433            _ = &mut connect => {}
434            _ = tokio::time::sleep(Duration::from_millis(500)) => {
435                let _ = stop_tx.try_send(());
436                let _ = (&mut connect).await;
437            }
438        }
439        tokio::time::sleep(Duration::from_millis(400)).await;
440        let got = collected.lock().unwrap().clone();
441        assert!(got.contains(&"AAPL".to_string()), "must resubscribe the survivor: {got:?}");
442        assert!(
443            !got.contains(&"MSFT".to_string()),
444            "must NOT resurrect the unsubscribed symbol: {got:?}"
445        );
446    }
