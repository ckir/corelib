1//! Diagnostics: env-gated native-thread flood hook to VALIDATE TSFN inbound delivery under
2//! GC reentrancy (the top audit residual). Production no-op unless CORELIB_DIAG_FLOOD=1.
3use napi::bindgen_prelude::Unknown;
4use napi::threadsafe_function::{ThreadsafeFunction, ThreadsafeFunctionCallMode};
5use napi::Status;
6use napi_derive::napi;
7
8/// Floods `on_event` with `count` synthetic JSON-string events from a dedicated native thread.
9/// Mirrors the real `on_market_event` interface (`ThreadsafeFunction<String>`, error-first
10/// delivery → JS sees `(null, jsonString)`). Gated behind `CORELIB_DIAG_FLOOD=1` so the symbol
11/// is always exported (stable index.d.ts) but does nothing in production.
12#[napi]
13pub fn napi_trigger_diagnostic_flood(
14    count: u32,
15    on_event: ThreadsafeFunction<String>,
16) -> napi::Result<()> {
17    if std::env::var("CORELIB_DIAG_FLOOD").unwrap_or_default() != "1" {
18        return Ok(()); // production no-op
19    }
20    std::thread::spawn(move || {
21        // Blocking delivery guarantees backpressure so no events are dropped. The call result is
22        // intentionally ignored: on teardown the TSFN may be Closing/InvalidArg, and we must never
23        // unwrap/panic on those non-Ok statuses.
24        for i in 0..count {
25            let ev = format!("{{\"seq\":{i}}}"); // synthetic event as a JSON string
26            let _ = on_event.call(Ok(ev), ThreadsafeFunctionCallMode::Blocking);
27        }
28    });
29    Ok(())
30}
31
32fn loadgen_enabled() -> bool {
33    std::env::var("CORELIB_LOADGEN").unwrap_or_default() == "1"
34}

36/// Parametric in-process synthetic load generator (Epic 5 audit instrument). Emits synthetic JSON
37/// ticks straight from a native thread → TSFN (no socket), supporting steady and bursty modes.
38/// Gated behind `CORELIB_LOADGEN=1` so the symbol is always exported but does nothing in production.
39/// TEST-ONLY: this must never be wired onto the real streaming pump.
40#[napi]
41pub fn napi_load_generator(
42    rate_per_sec: u32,
43    duration_ms: u32,
44    bursty: bool, // false = steady; true = recurrent spike→idle→gc-window cycles
45    on_event: ThreadsafeFunction<String, Unknown<'static>, String, Status, true, false, 1024>,
46) -> napi::Result<()> {
47    if !loadgen_enabled() {
48        return Ok(()); // production no-op
49    }
50    std::thread::spawn(move || {
51        let deadline =
52            std::time::Instant::now() + std::time::Duration::from_millis(duration_ms as u64);
53        let mut seq: u64 = 0;
54        while std::time::Instant::now() < deadline {
55            if bursty {
56                // spike: emit one second's worth fast, then idle 200ms to let the loop drain + GC
57                for _ in 0..rate_per_sec {
58                    seq += 1;
59                    crate::observability::latency::mark_sent(seq);
60                    let _ = on_event.call(
61                        Ok(synthetic_tick(seq)),
62                        ThreadsafeFunctionCallMode::Blocking,
63                    );
64                }
65                std::thread::sleep(std::time::Duration::from_millis(200));
66            } else {
67                // steady: pace to rate_per_sec
68                let per = std::time::Duration::from_micros(1_000_000 / rate_per_sec.max(1) as u64);
69                seq += 1;
70                crate::observability::latency::mark_sent(seq);
71                let _ = on_event.call(
72                    Ok(synthetic_tick(seq)),
73                    ThreadsafeFunctionCallMode::Blocking,
74                );
75                std::thread::sleep(per);
76            }
77        }
78    });
79    Ok(())
80}
81
82fn synthetic_tick(seq: u64) -> String {
83    // shaped like a real pricing payload but carrying a sequence id for loss/latency accounting
84    format!("{{\"seq\":{seq},\"symbol\":\"TEST\",\"price\":1.0,\"ts\":0}}")
85}
