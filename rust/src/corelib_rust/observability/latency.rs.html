1//! Test-only roundtrip-token latency accounting for the Epic 5 audit load generator.
2//!
3//! LOCK-FREE, ZERO-ALLOC. The load generator (`napi_load_generator`) calls `mark_sent(seq)` as it
4//! emits tick `seq`; JS calls `napi_latency_ack(seq)` immediately on receipt; elapsed micros are
5//! computed Rust-side against the SAME `Instant` epoch and pushed into a lock-free `ArrayQueue<u64>`.
6//! NEVER subtract Node's clock from Rust's. `seq` is a monotonic u64 indexing a fixed atomic-slot
7//! array (power-of-two mask = modulo); SLOTS far exceeds in-flight count so collisions are negligible
8//! for statistical latency.
9//!
10//! TEST-ONLY: this accounting must NEVER be wired onto the real production streaming pump. It exists
11//! solely to measure the synthetic load generator.
12use crossbeam_queue::ArrayQueue;
13use napi::threadsafe_function::{ThreadsafeFunction, ThreadsafeFunctionCallMode};
14use napi_derive::napi;
15use std::sync::atomic::{AtomicU64, Ordering};
16use std::sync::OnceLock;
17use std::time::Instant;
18
19const SLOTS: usize = 1 << 20; // 1,048,576
20static EPOCH: OnceLock<Instant> = OnceLock::new();
21static SENT_US: OnceLock<Box<[AtomicU64]>> = OnceLock::new(); // per-slot send time, micros-since-EPOCH (0 = empty)
22static SAMPLES: OnceLock<ArrayQueue<u64>> = OnceLock::new(); // elapsed micros
23
24fn epoch() -> &'static Instant {
25    EPOCH.get_or_init(Instant::now)
26}
27fn sent_us() -> &'static [AtomicU64] {
28    SENT_US.get_or_init(|| (0..SLOTS).map(|_| AtomicU64::new(0)).collect())
29}
30fn samples() -> &'static ArrayQueue<u64> {
31    SAMPLES.get_or_init(|| ArrayQueue::new(65536))
32}
33
34pub fn mark_sent(seq: u64) {
35    let us = epoch().elapsed().as_micros() as u64;
36    sent_us()[(seq as usize) & (SLOTS - 1)].store(us.max(1), Ordering::Relaxed); // never store 0 (= empty sentinel)
37}
38
39/// JS calls this immediately on receipt of tick `seq`. All timing is Rust-side vs the SAME EPOCH `Instant`.
40#[napi]
41pub fn napi_latency_ack(seq: i64) {
42    let now = epoch().elapsed().as_micros() as u64;
43    let start = sent_us()[(seq as usize) & (SLOTS - 1)].swap(0, Ordering::Relaxed);
44    if start != 0 {
45        let _ = samples().force_push(now.saturating_sub(start));
46    }
47}
48
49/// Drain samples to JS for percentile computation (env-gated, like the flight dump).
50#[napi]
51pub fn napi_latency_drain(on_sample: ThreadsafeFunction<f64>) -> napi::Result<()> {
52    if std::env::var("CORELIB_LOADGEN").unwrap_or_default() != "1" {
53        return Ok(());
54    }
55    while let Some(us) = samples().pop() {
56        let _ = on_sample.call(Ok(us as f64), ThreadsafeFunctionCallMode::NonBlocking);
57    }
58    Ok(())
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64
65    #[test]
66    fn mark_ack_roundtrip_records_a_sample() {
67        // pick a seq unlikely to collide with other tests
68        let seq = 123_456_u64;
69        mark_sent(seq);
70        napi_latency_ack(seq as i64);
71        // a sample must now be drainable; pop it directly (drain is env-gated)
72        assert!(samples().pop().is_some(), "ack after mark_sent must record a sample");
73    }
74
75    #[test]
76    fn ack_without_mark_records_nothing() {
77        let seq = 987_654_u64;
78        // ensure slot empty
79        sent_us()[(seq as usize) & (SLOTS - 1)].store(0, Ordering::Relaxed);
80        let before = samples().len();
81        napi_latency_ack(seq as i64);
82        assert_eq!(samples().len(), before, "ack without mark_sent must not record");
83    }
84}
