10struct FieldCollector {
11    message: String,
12    fields: String,
13}
14impl Visit for FieldCollector {
15    fn record_debug(&mut self, field: &Field, value: &dyn std::fmt::Debug) {
16        if field.name() == "message" {
17            let _ = write!(self.message, "{value:?}");
18        } else {
19            let _ = write!(self.fields, "{}={:?} ", field.name(), value);
20        }
21    }
22}

24fn level_str(l: &Level) -> &'static str {
25    match *l {
26        Level::TRACE => "TRACE",
27        Level::DEBUG => "DEBUG",
28        Level::INFO => "INFO",
29        Level::WARN => "WARN",
30        Level::ERROR => "ERROR",
31    }
32}

34/// tracing Layer that captures every event into the lock-free ring buffer.
35pub struct FlightLayer;
36impl<S: Subscriber> Layer<S> for FlightLayer {
37    fn on_event(&self, event: &Event<'_>, _ctx: Context<'_, S>) {
38        let meta = event.metadata();
39        let mut c = FieldCollector {
40            message: String::new(),
41            fields: String::new(),
42        };
43        event.record(&mut c);
44        record(FlightEvent {
45            ts_ms: now_ms(),
46            level: level_str(meta.level()),
47            target: meta.target(), // &'static str — no allocation
48            message: c.message,
49            fields: c.fields,
50        });
51    }
52}

54static INIT: Once = Once::new();
55
56/// Idempotent: installs the FlightLayer subscriber (+ EnvFilter) and a panic
57/// hook that dumps the ring to stderr. Safe to call on every host start.
58pub fn init_flight_recorder() {
59    INIT.call_once(|| {
60        // Default keeps the per-TICK firehose OFF (it lives under the distinct
61        // `corelib_rust::stream::tick` target — see later task) so the ring isn't washed
62        // out in ~1.6s under load; connect/reconnect/subscribe/error stay at debug
63        // in the ring. Opt into ticks with CORELIB_LOG="corelib_rust::stream::tick=trace".
64        let filter = EnvFilter::try_from_env("CORELIB_LOG").unwrap_or_else(|_| {
65            EnvFilter::new("corelib_rust=trace,corelib_rust::stream::tick=off")
66        });
67        // try_init: don't panic if a global subscriber already exists (tests/host re-entry).
68        let _ = tracing_subscriber::registry()
69            .with(filter)
70            .with(FlightLayer)
71            .try_init();
72        install_panic_dump_hook();
73    });
74}
75
76fn install_panic_dump_hook() {
77    let prev = std::panic::take_hook();
78    std::panic::set_hook(Box::new(move |info| {
79        // Pop per-event straight to stderr (no Vec/batch built during unwind →
80        // minimal alloc, can't OOM-double-panic). ArrayQueue can't poison → safe.
81        use std::io::Write as _;
82        let mut err = std::io::stderr().lock();
83        let _ = writeln!(err, "--- corelib flight-log dump (panic) ---");
84        let r = super::ring_buffer::ring();
85        while let Some(ev) = r.pop() {
86            let _ = writeln!(
87                err,
88                "[flight] {} [{}] {} {} {}",
89                ev.ts_ms, ev.level, ev.target, ev.message, ev.fields
90            );
91        }
92        let _ = err.flush();
93        prev(info);
94    }));
95}
